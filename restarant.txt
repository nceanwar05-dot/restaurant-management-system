#(practical)(restaurant management system)
import tkinter as tk
from tkinter import ttk
from tkinter import messagebox
import sqlite3

MENU = {
    "Pizza": 250,
    "Burger": 120,
    "Pasta": 180,
    "Coffee": 80,
    "Tea": 40,
    "Cold Drink": 60
}

class RestaurantApp:
    def __init__(self, root):
        self.root = root
        self.root.title("Restaurant Billing System")
        self.root.geometry("900x650")
        self.root.config(bg="#f8f9fa")

        self.setup_ui()

    def setup_ui(self):
        # Title Frame
        self.title_frame = tk.Frame(self.root, bg="#343a40")
        self.title_frame.pack(pady=10, fill="x")

        # Menu Management Buttons
        self.menu_btn_frame = tk.Frame(self.title_frame, bg="#343a40")
        self.menu_btn_frame.pack(side="left", padx=10)

        self.add_item_btn = tk.Button(self.menu_btn_frame, text="+ Add Item",
                                      bg="#17a2b8", fg="white", font=("Arial", 8), bd=0,state="disabled")
        self.add_item_btn.pack(side="left", padx=2)

        self.delete_item_btn = tk.Button(self.menu_btn_frame, text="- Del Item",
                                         bg="#dc3545", fg="white", font=("Arial", 8), bd=0, state="disabled")
        self.delete_item_btn.pack(side="left", padx=2)

        # Main input frame
        self.input_frame = tk.Frame(self.root, bg="#f8f9fa")
        self.input_frame.pack(pady=10)

        # Item selection
        tk.Label(self.input_frame, text="Select Item:", bg="#f8f9fa").grid(row=0, column=0, padx=5, pady=5)
        self.item_combo = ttk.Combobox(self.input_frame, values=list(MENU.keys()), state="disabled")
        self.item_combo.grid(row=0, column=1, padx=5, pady=5)
        self.item_combo.current(0)

        # Quantity
        tk.Label(self.input_frame, text="Quantity:", bg="#f8f9fa").grid(row=0, column=2, padx=5, pady=5)
        self.qty_entry = tk.Entry(self.input_frame)
        self.qty_entry.grid(row=0, column=3, padx=5, pady=5)

        # Action buttons
        self.add_btn = tk.Button(self.input_frame, text="Add Item", bg="#007bff", fg="white")
        self.add_btn.grid(row=0, column=4, padx=5, pady=5)

        self.delete_btn = tk.Button(self.input_frame, text="Delete Item", bg="#dc3545",
                                    fg="white")
        self.delete_btn.grid(row=0, column=5, padx=5, pady=5)

        # Order treeview
        self.order_tree = ttk.Treeview(self.root, columns=("#", "Item", "Quantity", "Price"), show="headings")
        self.order_tree.heading("#", text="#")
        self.order_tree.heading("Item", text="Item")
        self.order_tree.heading("Quantity", text="Qty")
        self.order_tree.heading("Price", text="Price")
        self.order_tree.pack(pady=10, fill="x")

        # Bottom frame
        self.bottom_frame = tk.Frame(self.root, bg="#f8f9fa")
        self.bottom_frame.pack(pady=10)

        self.total_var = tk.StringVar(value="Total: ₹0")
        self.total_label = tk.Label(self.bottom_frame, textvariable=self.total_var, font=("Arial", 14), bg="#f8f9fa",
                                    fg="#343a40")
        self.total_label.pack()

        # Action buttons
        self.btn_frame = tk.Frame(self.root, bg="#f8f9fa")
        self.btn_frame.pack(pady=10)

        self.clear_btn = tk.Button(self.btn_frame, text="Clear Order", bg="#ffc107",
                                   fg="black")
        self.clear_btn.grid(row=0, column=0, padx=10)

        self.bill_btn = tk.Button(self.btn_frame, text="Generate Bill", bg="#28a745",
                                  fg="white")
        self.bill_btn.grid(row=0, column=1, padx=10)

        # Sales report button (bottom right)
        self.sales_btn = tk.Button(self.root, text="Daily Sales", bg="#6c757d",
                                   fg="white")
        self.sales_btn.place(relx=0.95, rely=0.95, anchor="se")

        self.login_btn = tk.Button(self.root, text="Master Login",
                                   bg="#6c757d", fg="white", command=self.show_login)
        self.login_btn.place(relx=0.05, rely=0.95, anchor="sw")

        self.login_btn = tk.Button(self.root, text="Register",
                                   bg="#6c757d", fg="white", command=self.register)
        self.login_btn.place(relx=0.50, rely=0.95, anchor="sw")

    def register(self):
        self.login_window = tk.Toplevel(self.root)
        self.login_window.title("Register")
        self.login_window.geometry("300x150")

        tk.Label(self.login_window, text="Enter Password To Be Registered:").pack(pady=10)

        self.pr_entry = tk.Entry(self.login_window, show="*")
        self.pr_entry.pack(pady=5)

        tk.Button(self.login_window, text="Register", command=self.reg).pack(pady=10)
        self.pr_entry.focus_set()

    def reg(self):
        self.newname = self.pr_entry.get()
        conn = sqlite3.connect("users.db")
        cursor = conn.cursor()
        cursor.execute('''
        CREATE TABLE IF NOT EXISTS login (
            password TEXT
        )
        ''')
        cursor.execute("INSERT INTO login (password) VALUES (?)", (self.newname,))
        conn.commit()
        cursor.close()
        conn.close()



    def show_login(self):
        self.login_window = tk.Toplevel(self.root)
        self.login_window.title("Master Login")
        self.login_window.geometry("300x150")

        tk.Label(self.login_window, text="Enter Master Password:").pack(pady=10)

        self.pw_entry = tk.Entry(self.login_window, show="*")
        self.pw_entry.pack(pady=5)

        tk.Button(self.login_window, text="Login", command=self.check_login).pack(pady=10)
        self.pw_entry.focus_set()


    def check_login(self):
        self.password = self.pw_entry.get()
        self.conn = sqlite3.connect('users.db')
        self.cursor = self.conn.cursor()
        self.cursor.execute("SELECT * FROM login")
        rows = self.cursor.fetchall()

        counter = 0
        for row in rows:
            if self.password == row[0] :
                counter=1
            else:
                pass

        if counter == 1:
            messagebox.showinfo("success", "success")
            self.add_item_btn.config(state="normal")
            self.delete_item_btn.config(state="normal")
            self.item_combo.config(state="normal")
            self.login_window.destroy()
        else:
            messagebox.showerror("error", "error")


root = tk.Tk()
app = RestaurantApp(root)
root.mainloop()
